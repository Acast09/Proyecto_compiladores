topology TwoRouters {

    network LAN_A 192.168.1.0/24
    network TRANSIT 10.0.0.0/30
    network LAN_B 192.168.2.0/24

    router R1 {
        interface eth0 192.168.1.1/24 network LAN_A
        interface eth1 10.0.0.1/30 network TRANSIT

        route 192.168.2.0/24 via 10.0.0.2
    }

    router R2 {
        interface eth0 10.0.0.2/30 network TRANSIT
        interface eth1 192.168.2.1/24 network LAN_B

        route 192.168.1.0/24 via 10.0.0.1
    }

    connect R1.eth1 R2.eth0

}
